3.Write a function formatFollowersCount(count) that takes a number and returns a formatted string like Instagram: 1500 as '1.5K', 1200000 as '1.2M', and numbers below 1000 as-is. Add clear comments and use proper indentation.
#include <stdio.h>

#include <stdio.h>
#include <string.h>

void formatFollowersCount(int count, char output[]) {
    if (count < 1000) {
        sprintf(output, "%d", count);
    }
    else if (count < 1000000) {
        sprintf(output, "%.1fK", count / 1000.0);
    }
    else {
        sprintf(output, "%.1fM", count / 1000000.0);
    }
}

int main() {
    int count1 = 1500;
    int count2 = 1200000;
    int count3 = 850;

    char output[20];

    formatFollowersCount(count1, output);
    printf("%d = %s\n", count1, output);

    formatFollowersCount(count2, output);
    printf("%d = %s\n", count2, output);

    formatFollowersCount(count3, output);
    printf("%d = %s\n", count3, output);

    return 0;
}
