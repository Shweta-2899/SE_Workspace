2.Write a function swapPlaylistCounts(int *a, int *b) that swaps the number of songs in two Spotify playlists using pointers, then call the function in main and print 
the swapped values.

#include <stdio.h>
#include <conio.h>

void swapPlaylistCounts(int *a, int *b)
{
  int temp=*a;
  *a=*b;
  *b=temp;
};

void main(){
    
    int playlists1=10;
    int playlists2=20;
    printf("Before swapping\n");
    printf("%d\n",playlists1);
    printf("%d\n",playlists2);
    
    swapPlaylistCounts(&playlists1,&playlists2);
    printf("After swapping\n");
    printf("%d\n",playlists1);
    printf("%d\n",playlists2);
    getch();
}

4.Create a function incrementFollowers(int *followers, int n) that increases each follower count in an array (representing Instagram followers 
for 5 friends) by 100 using pointer arithmetic, then print the updated counts.

#include <stdio.h>
#include <conio.h>
 void incrementFollowers(int *followers, int n)
 {
     for(int i=0;i<n;i++)
     {
        *(followers+i) +=100;
     }
 }
void main(){
    int followers[5]={300,255,356,400,189};
    incrementFollowers(followers, 5);

}
