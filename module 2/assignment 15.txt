1.Create a console-based app called Music Listening Logger that allows users to enter the number of minutes they listened to music each day for a week and stores this data in an array.

2.Add a menu-driven interface to your Music Listening Logger so users can choose to log new listening minutes, view their weekly summary, or exit the app.<br><br><em><strong>Hint:</strong> Use a loop to repeatedly show the menu until the user selects exit.</em>

3.Implement file handling in your Music Listening Logger so that when the user logs their daily listening minutes, the data is saved to a file named music_log.txt for persistence.

4.Add a feature to your app that reads the saved music_log.txt file and generates a weekly report showing the total, average, and highest listening minutes in the week.

5.Refactor your Music Listening Logger to allow users to reset their weekly data by clearing the array and deleting the contents of music_log.txt.<br><br><em><strong>Constraint:</strong> Make sure the reset option is available in the menu and asks for confirmation before deleting data.</em>

#include <stdio.h>

int main() {
    int minutes[7] = {0};

    int choice;
    int i;

    do {
        printf("Music Listening Logger\n");
        printf("1. Log new listening minutes\n");
        printf("2. View weekly summary\n");
        printf("3. Reset weekly data\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice == 1) {

            printf("\nEnter the number of minutes you listened each day:\n");

            for (i = 0; i < 7; i++) {
                printf("Day %d: ", i + 1);
                scanf("%d", &minutes[i]);
            }

            FILE *file;

            file = fopen("music_log.txt", "w");

            if (file == NULL) {
                printf("Error opening file.\n");
            } else {

                for (i = 0; i < 7; i++) {
                    fprintf(file, "%d\n", minutes[i]);
                }

                fclose(file);

                printf("Data saved successfully.\n");
            }
        }

        else if (choice == 2) {

            FILE *file;
            int total = 0;
            int highest = 0;
            float average;

            file = fopen("music_log.txt", "r");

            if (file == NULL) {
                printf("No saved data found.\n");
            } else {

                for (i = 0; i < 7; i++) {
                    fscanf(file, "%d", &minutes[i]);
                }

                fclose(file);

                for (i = 0; i < 7; i++) {
                    total = total + minutes[i];
                }

                highest = minutes[0];

                for (i = 1; i < 7; i++) {
                    if (minutes[i] > highest) {
                        highest = minutes[i];
                    }
                }

                average = total / 7.0;

                printf("\n===== Weekly Report =====\n");

                for (i = 0; i < 7; i++) {
                    printf("Day %d: %d minutes\n",
                           i + 1, minutes[i]);
                }

                printf("Total: %d minutes\n", total);
                printf("Average: %.2f minutes\n", average);
                printf("Highest: %d minutes\n", highest);
            }
        }

        else if (choice == 3) {

            char confirm;
            FILE *file;

            printf("\nAre you sure you want to reset your weekly data? (y/n): ");
            scanf(" %c", &confirm);

            if (confirm == 'y' || confirm == 'Y') {

                for (i = 0; i < 7; i++) {
                    minutes[i] = 0;
                }

                file = fopen("music_log.txt", "w");

                if (file == NULL) {
                    printf("Error clearing the file.\n");
                } else {

                    fclose(file);

                    printf("Weekly data has been reset successfully.\n");
                }

            } else {
                printf("Reset cancelled.\n");
            }
        }

        else if (choice == 4) {
            printf("\nThank you for using Music Listening Logger!\n");
        }

        else {
            printf("\nInvalid choice. Please try again.\n");
        }

    } while (choice != 4);

    return 0;
}
