1.Create a text file called my_fav_songs.txt and write the names of your 5 favorite songs into it using ofstream.

#include <iostream>
#include <fstream>
using namespace std;

int main() 
{
    ofstream file("my_fav_songs.txt");

    file << "Seven by JungKook" << endl;
    file << "Cheap Thrills by Sia" << endl;
    file << "Show me Love by Tyla" << endl;
    file << "Fate of Ophelia by Taylor Swift" << endl;
    file << "Dandelions by Ruth B" << endl;

    file.close();

    cout << "Songs written to my_fav_songs.txt" << endl;

    return 0;
}

2.Read all song names from my_fav_songs.txt using ifstream and display each song on a new line in the console.

#include <iostream>
#include <fstream>
using namespace std;

int main() 
{
    ifstream file("my_fav_songs.txt");
    string song;

    while (getline(file, song)) 
    {
        cout << song << endl;
    }

    file.close();

    return 0;
}

4. Build a Flipkart-style wishlist tracker: ask the user to enter 3 product names and prices, save them to a file called wishlist.txt, then read the file and display each product with its price.

#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() 
{
    string product;
    double price;

    ofstream outFile("wishlist.txt");

    for (int i = 1; i <= 3; i++) 
    {
        cout << "Enter product " << i << " name: ";
        getline(cin, product);

        cout << "Enter price: ₹";
        cin >> price;
        cin.ignore();

        outFile << product << endl;
        outFile << price << endl;
    }

    outFile.close();

    ifstream inFile("wishlist.txt");

    cout << "My Wishlist" << endl;

    for (int i = 1; i <= 3; i++) 
    {
        getline(inFile, product);
        inFile >> price;
        inFile.ignore();

        cout << "Product: " << product << endl;
        cout << "Price: ₹" << price << endl;
        cout << endl;
    }

    inFile.close();

    return 0;
}

5.Given a file called insta_followers.txt containing Instagram usernames (one per line), write a program to count and display the total number of followers listed in the file.<br><br><em><strong>Constraint:</strong> Do not use any array or vector to store the names—just count as you read.</em>

#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() 
{
    ifstream file("insta_followers.txt");

    string username;
    int count = 300 ;

    while (getline(file, username)) 
    {
        count++;
    }

    cout << "Total followers: " << count << endl;

    file.close();

    return 0;
}
