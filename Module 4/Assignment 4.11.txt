5.Add a theme switcher button to your page that toggles between a light and dark background using Bootstrap classes, and updates the text color for readability.<br><br><em><strong>Hint:</strong> Use JavaScript to toggle 'bg-light'/'bg-dark' and 'text-dark'/'text-light' classes on the main container.</em>

<!doctype html>
<html lang="en">
<head>
	<meta charset="utf-8">
	<meta name="viewport" content="width=device-width, initial-scale=1">
	<title>Book Tickets</title>
	<link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/css/bootstrap.min.css" rel="stylesheet">
	<link href="styles.css" rel="stylesheet">
</head>
<body>
	<main id="theme-container" class="min-vh-100 bg-light text-dark">
		<div class="container min-vh-100 d-flex flex-column py-4">
			<header class="d-flex justify-content-end">
				<button id="theme-toggle" type="button" class="btn btn-outline-secondary" aria-pressed="false">Switch to dark mode</button>
			</header>
			<section class="flex-grow-1 d-flex flex-column justify-content-center align-items-center text-center py-5">
				<p class="small fw-bold text-uppercase text-danger mb-2">Make it a night to remember</p>
				<h1 class="h2 fw-bold mb-3">Your next great experience</h1>
				<p class="mb-4">Find a show, pick your seats, and enjoy the moment.</p>
				<button type="button" class="btn btn-primary btn-lg">Book Ticket</button>
			</section>
		</div>
	</main>
	<script>
		const themeContainer = document.getElementById('theme-container');
		const themeToggle = document.getElementById('theme-toggle');

		themeToggle.addEventListener('click', () => {
			const useDarkTheme = themeContainer.classList.contains('bg-light');
			themeContainer.classList.toggle('bg-light', !useDarkTheme);
			themeContainer.classList.toggle('bg-dark', useDarkTheme);
			themeContainer.classList.toggle('text-dark', !useDarkTheme);
			themeContainer.classList.toggle('text-light', useDarkTheme);
			themeToggle.setAttribute('aria-pressed', String(useDarkTheme));
			themeToggle.textContent = useDarkTheme ? 'Switch to light mode' : 'Switch to dark mode';
		});
	</script>
</body>
</html>
