	<script>
		const musicLinksForm = document.getElementById('musicLinksForm');
		const previewCards = document.getElementById('preview-cards');

		const platforms = {
			spotify: { name: 'Spotify', icon: 'S', cardClass: 'spotify-card', iconClass: 'spotify-icon' },
			youtube: { name: 'YouTube Music', icon: '▶', cardClass: 'youtube-card', iconClass: 'youtube-icon' },
			apple: { name: 'Apple Music', icon: 'A', cardClass: 'apple-card', iconClass: 'apple-icon' }
		};

		function getHandle(link) {
			try {
				const url = new URL(link);
				const parts = url.pathname.split('/').filter(Boolean);
				return parts.at(-1) || url.hostname;
			} catch (error) {
				return link;
			}
		}

		musicLinksForm.addEventListener('submit', (event) => {
			event.preventDefault();
			const formData = new FormData(musicLinksForm);
			previewCards.replaceChildren();
			let linkCount = 0;

			Object.entries(platforms).forEach(([key, platform]) => {
				const link = formData.get(key).trim();
				if (!link) {
					return;
				}

				linkCount += 1;
				const card = document.createElement('article');
				card.className = `preview-card ${platform.cardClass}`;

				const icon = document.createElement('span');
				icon.className = `card-icon ${platform.iconClass}`;
				icon.setAttribute('aria-hidden', 'true');
				icon.textContent = platform.icon;

				const copy = document.createElement('div');
				copy.className = 'card-copy';
				const name = document.createElement('strong');
				name.textContent = platform.name;
				const handle = document.createElement('span');
				handle.textContent = getHandle(link);
				copy.append(name, handle);

				const visit = document.createElement('a');
				visit.className = 'visit-button';
				visit.href = link;
				visit.target = '_blank';
				visit.rel = 'noopener noreferrer';
				visit.textContent = 'Visit';

				card.append(icon, copy, visit);
				previewCards.append(card);
			});

			if (!linkCount) {
				previewCards.innerHTML = '<p class="empty-state">Add at least one profile URL to create a preview.</p>';
			}
		});
	</script>
