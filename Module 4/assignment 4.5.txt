5.Implement a simple light/dark mode switch for a BookMyShow movie detail page using internal CSS and a toggle button. When the button is clicked, switch between light and dark color themes for the page.<br><br><em><strong>Hint:</strong> Use JavaScript to add or remove a 'dark-mode' class on the <body> and define the color themes in CSS.</em>

<!DOCTYPE html>
<html lang="en">
<head>
	<meta charset="UTF-8">
	<meta name="viewport" content="width=device-width, initial-scale=1.0">
	<title>BookMyShow - Movie Details</title>
	<style>
		:root {
			--page-bg: #f4f4f5;
			--surface: #ffffff;
			--text: #202124;
			--muted-text: #666666;
			--accent: #e51b23;
			--border: #dddddd;
		}

		body.dark-mode {
			--page-bg: #171717;
			--surface: #252525;
			--text: #f5f5f5;
			--muted-text: #bdbdbd;
			--accent: #ff5b61;
			--border: #444444;
		}

		* {
			box-sizing: border-box;
		}

		body {
			margin: 0;
			background-color: var(--page-bg);
			color: var(--text);
			font-family: Arial, sans-serif;
			transition: background-color 0.2s ease, color 0.2s ease;
		}

		header {
			display: flex;
			align-items: center;
			justify-content: space-between;
			padding: 18px 8%;
			background-color: var(--surface);
			border-bottom: 1px solid var(--border);
		}

		header h1 {
			margin: 0;
			color: var(--accent);
			font-size: 1.6rem;
		}

		button {
			padding: 10px 16px;
			border: 0;
			border-radius: 4px;
			background-color: var(--accent);
			color: #ffffff;
			cursor: pointer;
			font-size: 0.95rem;
		}

		main {
			max-width: 900px;
			margin: 40px auto;
			padding: 0 24px;
		}

		.movie-card {
			display: grid;
			grid-template-columns: 220px 1fr;
			gap: 28px;
			padding: 24px;
			background-color: var(--surface);
			border: 1px solid var(--border);
			border-radius: 8px;
		}

		.poster {
			height: 330px;
			overflow: hidden;
			border-radius: 6px;
			background-color: #333333;
		}

		.poster img {
			display: block;
			width: 100%;
			height: 100%;
			object-fit: cover;
		}

		.movie-card h2 {
			margin-top: 0;
			font-size: 2rem;
		}

			}

			.movie-card {
				grid-template-columns: 1fr;
			}

			.poster {
				height: 360px;
			}

<body>

				<button class="book-button" type="button">Book Tickets</button>
			</div>
		</article>
	</main>

	<script>
		const themeToggle = document.getElementById('theme-toggle');

		themeToggle.addEventListener('click', () => {
			const darkModeEnabled = document.body.classList.toggle('dark-mode');
			themeToggle.textContent = darkModeEnabled ? 'Light Mode' : 'Dark Mode';
			themeToggle.setAttribute('aria-pressed', darkModeEnabled);
		});
	</script>
</body>
